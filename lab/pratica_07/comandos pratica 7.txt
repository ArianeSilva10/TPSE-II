#!/bin/bash

# Exportar o canal PWM
echo 0 > /sys/class/pwm/pwmchip0/export

# Configurar o período (20 ms)
echo 20000000 > /sys/class/pwm/pwmchip0/pwm0/period

# Habilitar o PWM
echo 1 > /sys/class/pwm/pwmchip0/pwm0/enable

# Mover o servo para 0°
echo 1000000 > /sys/class/pwm/pwmchip0/pwm0/duty_cycle
sleep 1

# Mover o servo para 90°
echo 1500000 > /sys/class/pwm/pwmchip0/pwm0/duty_cycle
sleep 1

# Mover o servo para 180°
echo 2000000 > /sys/class/pwm/pwmchip0/pwm0/duty_cycle
sleep 1

# Desabilitar o PWM
echo 0 > /sys/class/pwm/pwmchip0/pwm0/enable

# Liberar o canal PWM
echo 0 > /sys/class/pwm/pwmchip0/unexport



chmod +x pwm_led
chmod +x pwm_servo

# Exportando o PWM
echo 0 > /sys/class/pwm/pwmchip0/export

# Configurando o período (20 ms ou 50 Hz)
echo 20000000 > /sys/class/pwm/pwmchip0/pwm0/period

# Configurando o Duty Cycle para 1,5 ms (posição central do servo)
echo 1500000 > /sys/class/pwm/pwmchip0/pwm0/duty_cycle

# Habilitando o sinal PWM
echo 1 > /sys/class/pwm/pwmchip0/pwm0/enable

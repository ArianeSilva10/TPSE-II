#!/bin/bash

# Função para enviar comando
send_command() {
    i2cset -y 2 0x27 $1
    sleep 0.1
}

# Inicialização do LCD
echo "Inicializando LCD..."
send_command 0x30
send_command 0x30
send_command 0x30
send_command 0x20

# Configuração
send_command 0x28  # 4-bit mode, 2 linhas
send_command 0x0C  # Display ON, cursor OFF
send_command 0x06  # Cursor move direction
send_command 0x01  # Clear display
sleep 1

# Liga backlight
send_command 0x08

# Posiciona na primeira linha
send_command 0x80

# Escreve "TESTE"
send_command 0x54  # T
send_command 0x45  # E
send_command 0x53  # S
send_command 0x54  # T
send_command 0x45  # E

# Posiciona na segunda linha
send_command 0xC0

# Escreve "LCD I2C"
send_command 0x4C  # L
send_command 0x43  # C
send_command 0x44  # D
send_command 0x20  # space
send_command 0x49  # I
send_command 0x32  # 2
send_command 0x43  # C

echo "Teste concluído!"
